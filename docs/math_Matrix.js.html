/**
 * 
 * @param {type} m
 * @param {type} n
 * @returns {Array}
 */
Matrix.createMatrix = function(m, n)
{
    var mx = []
    for (var i=0;i<n;i++)
    {
        var buffer = new ArrayBuffer(m * 8);
        mx[i] = new Float64Array(buffer);
        for (var j=0;j<m;j++)
        {
            mx[i][j] = 0;
        }
    }
    return mx;
}

/**
 * 
 * @param {type} n
 * @returns {Float64Array}
 */
Matrix.createVector = function(n)
{
    var buffer = new ArrayBuffer(n * 8);
    var mx = new Float64Array(buffer);        
    for (var i=0;i<n;i++)
    {
        mx[i] = 0;
    }
    return mx;
}

/**
 * 
 * @param {type} a
 * @param {type} b
 * @param {type} res
 * @returns {undefined}
 */
Matrix.add = function(a, b, res)
{
    if (a.length>0 && (a[0].constructor===Array || a[0].constructor===Float64Array) )
    {    
        for (var j=0;j<a.length;j++)
        {
            for(var i=0;i<a[j].length;i++)
            {
                res[j][i] = a[j][i]+b[j][i];
            }
        }
    }
    else
    {
        for(var i=0;i<a.length;i++)
        {
            res[i] = a[i]+b[i];
        }        
    }

}

/**
 * 
 * @param {type} a
 * @param {type} b
 * @param {type} res
 * @returns {undefined}
 */
Matrix.sub = function(a, b, res)
{
    if (a.length>0 && (a[0].constructor===Array || a[0].constructor===Float64Array) )
    {    
        for (var j=0;j<a.length;j++)
        {
            for(var i=0;i<a[j].length;i++)
            {
                res[j][i] = a[j][i]-b[j][i];
            }
        }
    }
    else
    {
        for(var i=0;i<a.length;i++)
        {
            res[i] = a[i]-b[i];
        }        
    }

}

/**
 * 
 * @param {type} a
 * @param {type} b
 * @param {type} res
 * @param {type} diagw
 * @returns {undefined}
 */
Matrix.mul = function(a, b, res, diagw=0)
{
    var i, j, k;
    var na = a.length;
    var ma = na>0 && (a[0].constructor===Array || a[0].constructor===Float64Array) ? a[0].length : 1;
    var nb = b.length;
    var mb = nb>0 && (b[0].constructor===Array || b[0].constructor===Float64Array) ? b[0].length : 1;
    var colstart, colend;
    
    //if (na!=mb) throw Exception("produit interdit");
    if (ma>1)
    {
        // Matrice ma*na et mb*nb quelquonques
        if (mb>1)
        {
            for (i=0;i<ma;i++)
            {
                colstart = diagw==0?0:i-diagw;
                colend = diagw==0?nb:i+diagw+1;
                if (colstart<0) colstart=0;
                if (colend>=nb) colend=nb;
                for(j=colstart;j<colend;j++)
                {
                    res[j][i] = 0;
                    for (k=0;k<na;k++)
                    {
                        res[j][i] += a[k][i]*b[j][k];
                    }
                }
            }
        }
        else
        {
            var cnt = 0;
            for (i=0;i<ma;i++)
            {
                colstart = diagw==0?0:i-diagw;
                colend = diagw==0?nb:i+diagw+1;
                if (colstart<0) colstart=0;
                if (colend>=nb) colend=nb;
                cnt += colend-colstart+1;
                res[i] = 0;
                for(j=colstart;j<colend;j++)
                {
                    res[i] += a[j][i]*b[j];
                }
            }
        }
    }
    else 
    {
        for(j=0;j<nb;j++)
        {
            res[j][0] = 0;
            for (k=0;k<na;k++)
            {
                res[j][0] += a[k][0]*b[j][k];
            }
        }
    }

}

/**
 * 
 * @param {type} x
 * @returns {Number}
 */
Matrix.norm = function(x)
{
    var i, r;
    r = 0;
    for (i=0;i<x.length;i++)
    {
        r += x[i]*x[i];
    }
    return Math.sqrt(r);
    
}

/**
 * Résolution d'un système ax=b par la méthode de surrelaxation.
 * 
 * @param {type} a matrice du système
 * @param {type} b vecteur du système
 * @param {type} w coefficient de surrelaxation entre 0 et 1
 * @param {type} x matrice résultat. L'algorithme tient compte des valeurs 
 * initiales pour améliorer la convergence.
 * @param {type} r vecteur résidu, de même dimension que x.
 * @param {type} epsilon seuil optionnel pour détermination de la convergence. 
 * Défaut : 0.000001
 * @param {type} maxiter nombre maximum d'itérations avant abandon de la 
 * convergence. Défaut : 1000
 * @param {type} diagw largeur de la diagonale, pour optimiser les 
 * multiplicaions. Défaut : 0, pas d'optimisation.
 * @returns {Number} le nombre d'itérations pour atteindre la convergence.
 */
Matrix.sor = function(a, b, w, x, r, epsilon=0.000001, maxiter=1000, diagw=0)
{
    var i, j;
    var s, k;
    var nr;
    var na = a.length;
    var ma = na>0 && (a[0].constructor===Array || a[0].constructor===Float64Array) ? a[0].length : 1;
    
    if (ma!=na) throw Exception("matrice non carrée");
    
    for (i=0;i<ma;i++) r[i] = 1;

    k=0;
    while ((nr=Matrix.norm(r))>epsilon && k<maxiter)
    {
        k++;
        for (i=0;i<ma;i++)
        {
            s = 0;
            for (j=0;j<=i-1;j++)
            {
                s += a[j][i]*x[j];
            }
            for (j=i+1;j<na;j++)
            {
                s += a[j][i]*x[j];
            }
            x[i] = (1-w)*x[i]+w/a[i][i]*(b[i]-s);
        }
        Matrix.mul(a, x, r, diagw);
        Matrix.sub(r, b, r);
    }
    return k;
}
